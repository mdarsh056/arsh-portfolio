Module 1

Fundamentals, Compilation & Syntax

Core Concepts & Definitions

C Architecture & Compilation Pipeline: C is a middle-level, statically typed language. Source code (`.c`) transitions through 4 deterministic phases:

  • Preprocessing (`gcc -E`): Expands `#include`, removes comments, processes macros (`.i`).
  • Compilation (`gcc -S`): Converts preprocessed C to machine assembly code (`.s`).
  • Assembly (`gcc -c`): Converts assembly mnemonics into machine object code (`.o`).
  • Linking (`gcc`): Binds object files and standard C runtime libraries (`libc`) into an executable image (`.exe`).

Tokens & Storage: Keywords (32 reserved words), Identifiers (names of variables/functions), and Data Types (`char`: 1B, `int`: 4B, `float`: 4B, `double`: 8B, `void`: incomplete type).

Problem 1.1: ASCII Translation & Storage Inspector 
#include <stdio.h>

int main() {
    char ch;
    printf("Enter a character: ");
    scanf("%c", &ch);

    printf("Character: %c\n", ch);
    printf("ASCII Integer Value: %d\n", ch);
    printf("Memory Footprint: %zu byte(s)\n", sizeof(ch));
    return 0;
}
Problem 1.2: Primitive Memory Footprint & Limits

Module 2

Operators, Bitwise & Expressions

Core Concepts & Definitions

Operators formulate expressions evaluated by CPU ALU units based on Precedence (binding priority) and Associativity (left-to-right vs right-to-left).

  • Arithmetic & Assignment: `+`, `-`, `*`, `/`, `%` (only integer operands), shorthand: `+=`, `*=`.
  • Bitwise Operators: Direct binary bit manipulation: AND (`&`), OR (`|`), XOR (`^`), NOT (`~`), Left Shift (`<<` - multiplies by $2^k$), Right Shift (`>>` - divides by $2^k$).
  • Ternary Operator: Compact single-line conditional syntax `(condition) ? expr1 : expr2`.
Problem 2.1: Bitwise Odd/Even & Power-of-Two Verifier 
#include <stdio.h>

int main() {
    int n = 16;
    // LSB is 0 for even, 1 for odd
    if ((n & 1) == 0)
        printf("%d is Even\n", n);
    else
        printf("%d is Odd\n", n);

    // Power of 2 check: (n > 0) && (n & (n - 1)) == 0
    if (n > 0 && (n & (n - 1)) == 0)
        printf("%d is a power of 2\n", n);
    return 0;
}
Problem 2.2: Bitwise Value Swap without Temp Variable 
#include <stdio.h>

int main() {
    int a = 25, b = 70;
    printf("Before: a = %d, b = %d\n", a, b);
    a = a ^ b;
    b = a ^ b;
    a = a ^ b;
    printf("After:  a = %d, b = %d\n", a, b);
    return 0;
}
Module 3

Control Flow & Branching

Core Concepts & Definitions

Branching structures alter sequential CPU instruction execution via jump operations.

  • `if-else` / `else-if`: Evaluates boolean truth conditions (0 = False, Non-zero = True).
  • `switch-case`: Constant-expression dispatch table matching integers/characters. Requires `break` to prevent fall-through.
  • Jump Statements: `break` exits current loop/switch; `continue` skips to next cycle; `goto` executes unconditional jumps.
Problem 3.1: Tariff Slabs with Surcharge Engine (Level 2) 
#include <stdio.h>

int main() {
    float units = 250.0, bill = 0.0, surcharge = 0.0;
    if (units <= 100) bill = units * 4.0;
    else if (units <= 200) bill = (100 * 4.0) + (units - 100) * 6.0;
    else bill = (100 * 4.0) + (100 * 6.0) + (units - 200) * 8.5;

    if (bill > 400.0) surcharge = bill * 0.15;
    printf("Total Due: Rs. %.2f\n", bill + 50.0 + surcharge);
    return 0;
}
Problem 3.2: Switch-Case Arithmetic State Engine 
#include <stdio.h>

int main() {
    char op = '*';
    double a = 12.0, b = 4.0;
    switch(op) {
        case '+': printf("Sum: %.2f\n", a + b); break;
        case '-': printf("Diff: %.2f\n", a - b); break;
        case '*': printf("Product: %.2f\n", a * b); break;
        case '/': 
            if(b != 0) printf("Quotient: %.2f\n", a / b);
            else printf("Division by zero error!\n");
            break;
        default: printf("Unknown Operator\n");
    }
    return 0;
}
Module 4

Loops & Iterative Processing

Core Concepts & Definitions

  • `while` (Pre-tested): Evaluates condition prior to executing body. 0-iteration minimum.
  • `for` (Deterministic): Compact syntax combining initialization, condition test, and increment step.
  • `do-while` (Post-tested): Guaranteed minimum 1 execution cycle prior to testing condition.
Problem 4.1: Decimal Palindrome Number Verifier 
#include <stdio.h>

int main() {
    int n = 12321, original = n, rev = 0;
    while (n != 0) {
        rev = (rev * 10) + (n % 10);
        n /= 10;
    }
    if (original == rev) printf("%d is a Palindrome\n", original);
    else printf("%d is Not Palindrome\n", original);
    return 0;
}
Problem 4.2: Prime Number Sieve Check 
#include <stdio.h>
#include <stdbool.h>

int main() {
    int n = 37;
    bool isPrime = (n > 1);
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) { isPrime = false; break; }
    }
    printf("%d is %s\n", n, isPrime ? "Prime" : "Composite");
    return 0;
}
Module 5

Functions, Scope & Storage Classes

Core Concepts & Definitions

Functions isolate logic into discrete stack activation records.

  • Call by Value: Passes copies of argument data (caller variables remain untouched).
  • Call by Reference: Passes memory addresses via pointers, enabling direct mutation in caller scope.
  • Storage Classes: `auto` (default stack lifetime), `static` (persists value across lifetime of program), `extern` (global cross-file linkage), `register` (CPU register request).
Problem 5.1: Call by Reference Value Swapper 
#include <stdio.h>

void swapByRef(int *x, int *y) {
    int temp = *x;
    *x = *y;
    *y = temp;
}

int main() {
    int p = 10, q = 20;
    swapByRef(&p, &q);
    printf("p = %d, q = %d\n", p, q);
    return 0;
}
Problem 5.2: Stack Unwinding in Recursive GCD 
#include <stdio.h>

int gcd(int a, int b) {
    if (b == 0) return a; // Base case
    return gcd(b, a % b); // Tail recursion
}

int main() {
    printf("GCD(48, 18) = %d\n", gcd(48, 18));
    return 0;
}
Module 6

Arrays & String Manipulation

Core Concepts & Definitions

Arrays: Homogeneous contiguous blocks in memory accessed via index displacement: $\text{Addr}(A[i]) = \text{Base} + i \times \text{sizeof}(\text{Type})$.

Strings: Null-terminated character sequences (`char str[] = "text\0"`). The null byte `\0` signals string boundary to runtime routines.

Problem 6.1: Matrix Transpose in Memory 
#include <stdio.h>

int main() {
    int mat[2][3] = {{1, 2, 3}, {4, 5, 6}};
    int trans[3][2];

    for(int i = 0; i < 2; i++) {
        for(int j = 0; j < 3; j++) {
            trans[j][i] = mat[i][j];
        }
    }
    printf("Transposed Element [0][1]: %d\n", trans[0][1]);
    return 0;
}
Problem 6.2: String Inversion Without string.h 
#include <stdio.h>

int main() {
    char str[] = "KatiharEngineering";
    int len = 0;
    while(str[len] != '\0') len++;

    for(int i = 0, j = len - 1; i < j; i++, j--) {
        char tmp = str[i];
        str[i] = str[j];
        str[j] = tmp;
    }
    printf("Reversed: %s\n", str);
    return 0;
}
Module 7

Pointers & Memory Addressing

Core Concepts & Definitions

Pointers store raw virtual memory addresses. Key aspects include:

  • Pointer Arithmetic: Incrementing (`ptr++`) scales address by `sizeof(*ptr)` bytes.
  • Double Pointers (`**ptr`): Pointers storing the address of another pointer variable.
  • Pathological States: Dangling (points to deallocated space), Wild (uninitialized garbage address), NULL (points safely to `(void*)0`).
Problem 7.1: Array Traversal via Pointer Arithmetic 
#include <stdio.h>

int main() {
    int arr[] = {10, 20, 30, 40};
    int *ptr = arr; // Base address
    for (int i = 0; i < 4; i++) {
        printf("Index %d: Address %p | Value: %d\n", i, (void*)(ptr + i), *(ptr + i));
    }
    return 0;
}
Problem 7.2: Function Pointer Callback Dispatcher 
#include <stdio.h>

int add(int a, int b) { return a + b; }
int multiply(int a, int b) { return a * b; }

void executeOperation(int (*op)(int, int), int x, int y) {
    printf("Result: %d\n", op(x, y));
}

int main() {
    executeOperation(add, 5, 7);
    executeOperation(multiply, 5, 7);
    return 0;
}
Module 8

User-Defined Types: Structs, Unions & Enums

Core Concepts & Definitions

  • `struct`: Heterogeneous compound type. Total size $\ge \sum \text{member sizes}$ due to byte alignment/padding.
  • `union`: Shared overlapping memory space. Size matches the largest member only.
  • `typedef`: Creates convenient semantic aliases for complex type signatures.
Problem 8.1: Student Records via Struct Pointers 
#include <stdio.h>

typedef struct {
    int roll;
    char name[30];
    float marks;
} Student;

int main() {
    Student s1 = {101, "Arsh", 92.5};
    Student *sPtr = &s1;
    // Direct dereference via arrow operator (->)
    printf("Roll: %d | Name: %s | Marks: %.2f\n", sPtr->roll, sPtr->name, sPtr->marks);
    return 0;
}
Problem 8.2: Memory Overlap in C Unions 
#include <stdio.h>

union DataPacket {
    int intVal;
    float floatVal;
    char charVal;
};

int main() {
    union DataPacket packet;
    packet.intVal = 65;
    printf("Interpreted as int: %d\n", packet.intVal);
    printf("Interpreted as char (overlap): %c\n", packet.charVal);
    printf("Union Footprint: %zu bytes\n", sizeof(packet));
    return 0;
}
Module 9

Dynamic Memory Allocation (Heap Operations)

Core Concepts & Definitions

Allocates heap buffers at runtime when data sizes are variable or unknown at compile time.

  • `malloc(bytes)`: Allocates uninitialized memory buffer. Returns `void*` or `NULL` if exhausted.
  • `calloc(n, bytes)`: Allocates zero-initialized contiguous memory block.
  • `realloc(ptr, new_bytes)`: Resizes existing block while preserving data.
  • `free(ptr)`: Releases heap block back to OS pool (prevents memory leaks).
Problem 9.1: Dynamic Array Allocation & Sum 
#include <stdio.h>
#include <stdlib.h>

int main() {
    int n = 5;
    int *arr = (int *)malloc(n * sizeof(int));
    if (arr == NULL) return 1;

    for (int i = 0; i < n; i++) arr[i] = (i + 1) * 10;
    
    int sum = 0;
    for (int i = 0; i < n; i++) sum += arr[i];
    printf("Heap Array Sum: %d\n", sum);

    free(arr);
    arr = NULL; // Prevent dangling pointer
    return 0;
}
Problem 9.2: Dynamic Buffer Expansion via realloc 
#include <stdio.h>
#include <stdlib.h>

int main() {
    int *buffer = (int *)malloc(2 * sizeof(int));
    buffer[0] = 100; buffer[1] = 200;

    // Expand buffer from 2 to 4 integers
    int *temp = (int *)realloc(buffer, 4 * sizeof(int));
    if (temp != NULL) {
        buffer = temp;
        buffer[2] = 300; buffer[3] = 400;
        printf("Buffer[3]: %d\n", buffer[3]);
    }
    free(buffer);
    return 0;
}
Module 10

File I/O & Persistent Disk Streams

Core Concepts & Definitions

File handling links the program to permanent storage streams using the opaque `FILE *` handle.

  • Stream Modes: `"r"` (read), `"w"` (overwrite), `"a"` (append), `"rb"`/`"wb"` (binary).
  • Random Access: `fseek(fp, offset, origin)` positions cursor (`SEEK_SET`, `SEEK_CUR`, `SEEK_END`), `ftell(fp)` reports byte offset, `rewind(fp)` resets cursor.
Problem 10.1: Writing & Reading Text Records 
#include <stdio.h>

int main() {
    FILE *fp = fopen("output.txt", "w");
    if (!fp) return 1;
    fprintf(fp, "Token: %d Rate: %.2f\n", 404, 99.5);
    fclose(fp);

    fp = fopen("output.txt", "r");
    int token; float rate;
    fscanf(fp, "Token: %d Rate: %f", &token, &rate);
    printf("Read from disk -> Token: %d, Rate: %.2f\n", token, rate);
    fclose(fp);
    return 0;
}
Problem 10.2: Binary Struct Persistence with fread/fwrite 
#include <stdio.h>

typedef struct { int id; double balance; } Account;

int main() {
    Account acc1 = {9001, 45200.75};
    FILE *fp = fopen("acc.bin", "wb");
    fwrite(&acc1, sizeof(Account), 1, fp);
    fclose(fp);

    Account accLoaded;
    fp = fopen("acc.bin", "rb");
    fread(&accLoaded, sizeof(Account), 1, fp);
    printf("Loaded Account ID: %d | Bal: Rs. %.2f\n", accLoaded.id, accLoaded.balance);
    fclose(fp);
    return 0;
}
Module 11

Preprocessor Directives & CLI Mechanics

Core Concepts & Definitions

  • Macros (`#define`): Text substitutions executed before compilation. Function macros must use outer parentheses to prevent precedence bugs.
  • Conditional Compilation: `#ifdef`, `#ifndef`, `#endif` prevent multiple inclusions (Include Guards).
  • CLI Arguments (`argc`, `argv`): System parameters passed into `main(int argc, char *argv[])`.
Problem 11.1: Macro Hygiene with Guarded Expressions 
#include <stdio.h>

#define SQUARE(x) ((x) * (x))
#define MAX(a, b) (((a) > (b)) ? (a) : (b))

int main() {
    int res = SQUARE(3 + 2); // Evaluates properly to ((3 + 2) * (3 + 2)) = 25
    printf("Square Result: %d\n", res);
    printf("Max: %d\n", MAX(10, 25));
    return 0;
}
Problem 11.2: Command Line Argument Parser 
#include <stdio.h>
#include <stdlib.h>

int main(int argc, char *argv[]) {
    printf("Total Arguments: %d\n", argc);
    if (argc >= 3) {
        int val1 = atoi(argv[1]);
        int val2 = atoi(argv[2]);
        printf("Sum of CLI inputs: %d\n", val1 + val2);
    } else {
        printf("Usage: %s <num1> <num2>\n", argv[0]);
    }
    return 0;
}
